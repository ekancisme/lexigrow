  const groq = new Groq({ apiKey });
  const chatCompletion = await groq.chat.completions.create({
    messages: [
      { role: 'system', content: prompt },

nlp_processor.pyimport spacy, collections

def analyze_text(text):
    nlp = spacy.load("en_core_web_sm")
    doc = nlp(text)
    
    passive_voice_count = sum(1 for t in doc if t.dep_ == "auxpass")
    subordinate_clauses_count = sum(1 for t in doc if t.dep_ in ["advcl", "acl", "ccomp"])
    
    tokens = [t.text.lower() for t in doc if not t.is_punct and not t.is_space]
    def compute_mtld_dir(word_list):
        factor_count = 0.0
        start_idx = 0
        unique_seg = set()
        for idx, word in enumerate(word_list):
            unique_seg.add(word)
            seg_len = idx - start_idx + 1
            cur_ttr = len(unique_seg) / seg_len
            if cur_ttr < 0.72 and seg_len > 1:
                factor_count += 1.0
                unique_seg = set()
                start_idx = idx + 1
        return len(word_list) / factor_count if factor_count > 0 else len(word_list)
        
    mtld = round((compute_mtld_dir(tokens) + compute_mtld_dir(list(reversed(tokens)))) / 2.0, 1)

    freqs = collections.Counter(tokens)
    expected_unique = 0.0
    for word, c in freqs.items():
        ratio = 1.0
        for j in range(42):
            ratio *= (len(tokens) - c - j) / (len(tokens) - j)
        expected_unique += (1.0 - ratio)
    hdd = round(expected_unique / 42, 2)
    
    return { "passiveVoiceCount": passive_voice_count, "subordinateClausesCount": subordinate_clauses_count, "lexicalDiversity": { "mtld": mtld, "hdd": hdd } }

plagiarism.service.jsconst computeJaccard = (setA, setB) => {
  if (setA.size === 0 || setB.size === 0) return 0;
  let intersection = 0;
  for (const item of setA) {
    if (setB.has(item)) intersection++;
  }
  const union = setA.size + setB.size - intersection;
  return intersection / union;
};

Tóm Tắt Toàn Bộ Tiến Trình Chấm Bài

Quy trình tự động hóa khép kín từ lúc nhấn "Nộp bài" đến khi hệ thống lưu kết quả chấm vào Database.

1. Chuẩn hóa & Điều phối • Lọc bỏ thẻ HTML thô, nén khoảng trắng văn bản.
• Tải lịch sử điểm học sinh & gọi đồng thời (Promise.all) các engine phân tích để tối ưu hiệu năng CPU và độ trễ.
2. Chấm điểm & Xử lý NLP chuyên sâu • Groq LLM chấm điểm, chỉ ra ưu khuyết điểm và phát hiện nhồi từ.
• Subprocess Python (spaCy) phân tích bị động, mệnh đề phụ và đo MTLD / HD-D chính xác.
3. Kiểm soát Đạo văn & Nhận diện AI • Quét trigrams chéo giữa các học sinh bằng chỉ số Jaccard.
• Model RoBERTa Hugging Face phân loại văn bản AI, kết hợp bộ heuristics cục bộ thông minh về Marker Words & Burstiness.
4. Làm giàu tri thức & Lưu trữ • Phân tích từ mới, bổ sung IPA, nghĩa, sinh ví dụ đồng bộ ngữ cảnh.
• Cập nhật các bản ghi AIAnalysis, Vocabulary giúp sẵn sàng hiển thị kết quả cho Học sinh & Giáo viên.
